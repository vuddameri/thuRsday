AnovaModel.3 <- aov(PctEdu ~ Incomegroups, data=Dataset)
summary(AnovaModel.3)
with(Dataset, numSummary(PctEdu, groups=Incomegroups, statistics=c("mean", 
  "sd")))
local({
  .Pairs <- glht(AnovaModel.3, linfct = mcp(Incomegroups = "Tukey"))
  print(summary(.Pairs)) # pairwise tests
  print(confint(.Pairs, level=0.95)) # confidence intervals
  print(cld(.Pairs, level=0.05)) # compact letter display
  old.oma <- par(oma=c(0, 5, 0, 0))
  plot(confint(.Pairs))
  par(old.oma)
})
oneway.test(PctEdu ~ Incomegroups, data=Dataset) # Welch test
